public function getSubmissionById($submissionId)
{
    $stmt = $this->conn->prepare(
        "SELECT *
         FROM form_submission_values
         WHERE submission_id=?"
    );

    $stmt->bind_param("i", $submissionId);

    $stmt->execute();

    return $stmt->get_result()->fetch_all(MYSQLI_ASSOC);
}
public function updateFieldValue($submissionId,$fieldId,$value)
{
    $stmt = $this->conn->prepare(
        "UPDATE form_submission_values
         SET field_value=?
         WHERE submission_id=?
         AND field_id=?"
    );

    $stmt->bind_param(
        "sii",
        $value,
        $submissionId,
        $fieldId
    );

    return $stmt->execute();
}

public function edit()
{
    if (
        !isset($_GET['form_id']) ||
        !isset($_GET['submission_id'])
    ) {
        die("Invalid Request");
    }

    $formId = $_GET['form_id'];
    $submissionId = $_GET['submission_id'];

    // Get form details
    $form = $this->form->getForm($formId);

    // Get all fields
    $fields = $this->form->getFields($formId);

    // Get saved values
    $values = $this->submission->getSubmissionById($submissionId);

    // Convert values into an array
    $savedValues = [];

    foreach ($values as $value) {
        $savedValues[$value['field_id']] = $value['field_value'];
    }

    // Attach options and saved value to each field
    foreach ($fields as &$field) {

        if (
            $field['field_type'] == "select" ||
            $field['field_type'] == "radio" ||
            $field['field_type'] == "checkbox"
        ) {
            $field['options'] = $this->form->getOptions($field['id']);
        }

        $field['value'] = isset($savedValues[$field['id']])
            ? $savedValues[$field['id']]
            : "";
    }

    // Send data to Smarty
    $this->smarty->assign("form", $form);
    $this->smarty->assign("fields", $fields);
    $this->smarty->assign("submission_id", $submissionId);

    $this->smarty->display("dynamic_form.tpl");
}

public function update()
{
    if (
        !isset($_POST['form_id']) ||
        !isset($_POST['submission_id'])
    ) {
        die("Invalid Request");
    }

    $formId = $_POST['form_id'];

    $submissionId = $_POST['submission_id'];

    $fields = $this->form->getFields($formId);

    foreach ($fields as $field)
    {
        $fieldId = $field['id'];

        $fieldName = $field['field_name'];

        $fieldType = $field['field_type'];

        $value = "";

        // Checkbox
        if ($fieldType == "checkbox")
        {
            if (isset($_POST[$fieldName])) {
                $value = implode(",", $_POST[$fieldName]);
            }
        }

        // File Upload
        elseif ($fieldType == "file")
        {
            if (
                isset($_FILES[$fieldName]) &&
                $_FILES[$fieldName]['error'] == 0
            ) {

                $fileName =
                time()."_".$_FILES[$fieldName]['name'];

                move_uploaded_file(

                    $_FILES[$fieldName]['tmp_name'],

                    "../uploads/".$fileName

                );

                $value = $fileName;
            }
            else
            {
                // Keep old value
                $old =
                $this->submission
                     ->getSubmissionById($submissionId);

                foreach($old as $row)
                {
                    if($row['field_id']==$fieldId)
                    {
                        $value=$row['field_value'];
                        break;
                    }
                }
            }
        }

        // Normal Fields
        else
        {
            if(isset($_POST[$fieldName]))
            {
                $value = $_POST[$fieldName];
            }
        }

        $this->submission->updateFieldValue(

            $submissionId,

            $fieldId,

            $value

        );
    }

    header(
        "Location:index.php?action=list&form_id=".$formId
    );

    exit;
}
