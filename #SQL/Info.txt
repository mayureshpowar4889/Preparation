 <?php
session_start();

// Initialize dynamic system date tracks
if (!isset($_SESSION['year']))  { $_SESSION['year']  = (int)date('Y'); }
if (!isset($_SESSION['month'])) { $_SESSION['month'] = (int)date('n'); }
if (!isset($_SESSION['selected_day'])) { $_SESSION['selected_day'] = null; }

$alertMessage = "";
if (isset($_POST['action']) && $_POST['action'] === 'help') {
    $alertMessage = "<script>alert('this is calendar');</script>";
}

if (isset($_POST['select_day'])) {
    $_SESSION['selected_day'] = (int)$_POST['select_day'];
}

if (isset($_POST['action'])) {
    switch ($_POST['action']) {
        case 'prev_year':   $_SESSION['year']--; break;
        case 'next_year':   $_SESSION['year']++; break;
        case 'prev_month':
            $_SESSION['month']--;
            if ($_SESSION['month'] < 1) { $_SESSION['month'] = 12; $_SESSION['year']--; }
            break;
        case 'next_month':
            $_SESSION['month']++;
            if ($_SESSION['month'] > 12) { $_SESSION['month'] = 1; $_SESSION['year']++; }
            break;
        case 'today':
            $_SESSION['year']  = (int)date('Y');
            $_SESSION['month'] = (int)date('n');
            $_SESSION['selected_day'] = (int)date('j');
            break;
    }
}

$currentYear  = $_SESSION['year'];
$currentMonth = $_SESSION['month'];
$selectedDay  = $_SESSION['selected_day'];

$monthNames = [
    1 => "January", 2 => "February", 3 => "March", 4 => "April", 
    5 => "May", 6 => "June", 7 => "July", 8 => "August", 
    9 => "September", 10 => "October", 11 => "November", 12 => "December"
];

// Footer string calculations engine logic
$footerText = "Select date";
if ($selectedDay !== null) {
    $firstDayTimestamp = strtotime("$currentYear-$currentMonth-01");
    $totalDaysInMonth  = (int)date('t', $firstDayTimestamp);
    
    if ($selectedDay > $totalDaysInMonth) {
        $selectedDay = $totalDaysInMonth;
        $_SESSION['selected_day'] = $totalDaysInMonth;
    }

    $timeString = strtotime("$currentYear-$currentMonth-$selectedDay");
    $shortMonth = date('M', $timeString);
    $shortDayName = strtolower(date('D', $timeString));
    $footerText = "{$shortMonth} {$shortDayName} {$selectedDay}";
}

echo "<!DOCTYPE html>\n";
echo "<html lang='en'>\n";
echo "<head>\n";
echo "    <title> Calendar</title>\n";
echo "    <link rel='stylesheet' href='style.css'>\n";
echo "</head>\n";
echo "<body>\n";

if (!empty($alertMessage)) { echo $alertMessage; }

echo "<div class='calendars-wrapper'>\n";

if (!(isset($_POST['action']) && $_POST['action'] === "close_calendar")) {
    echo "    <div class='calendar-container'>\n";
    echo "        <form method='POST' action='index.php' style='margin:0; padding:0;'>\n";
    echo "            <table class='calendar-table'>\n";
    
    // Row 1
    echo "                <tr class='row-header'>\n";
    echo "                    <td class='btn-help'><button type='submit' name='action' value='help' data-info='Help button'>?</button></td>\n";
    echo "                    <td colspan='6' class='current-month-year'>{$monthNames[$currentMonth]} , {$currentYear}</td>\n";
    echo "                    <td class='btn-close'><button type='submit' name='action' value='close_calendar' data-info='Close calendar button'>✕</button></td>\n";
    echo "                </tr>\n";
    
    // Row 2
    echo "                <tr class='row-navigation'>\n";
    echo "                    <td class='nav-btn'><button type='submit' name='action' value='prev_year' data-info='Previous year button'>&lt;&lt;</button></td>\n";
    echo "                    <td class='nav-btn'><button type='submit' name='action' value='prev_month' data-info='Previous month button'>&lt;</button></td>\n";
    echo "                    <td colspan='4' class='nav-btn btn-today'><button type='submit' name='action' value='today' data-info='Today button'>Today</button></td>\n";
    echo "                    <td class='nav-btn'><button type='submit' name='action' value='next_month' data-info='Next month button'>&gt;</button></td>\n";
    echo "                    <td class='nav-btn'><button type='submit' name='action' value='next_year' data-info='Next year button'>&gt;&gt;</button></td>\n";
    echo "                </tr>\n";
    
    // Row 3
    echo "                <tr class='row-days-header'>\n";
    echo "                    <th class='wk-header'>WK</th>\n";
    echo "                    <th class='sun-header'>Sun</th>\n";
    echo "                    <th>Mon</th>\n";
    echo "                    <th>Tue</th>\n";
    echo "                    <th>Wed</th>\n";
    echo "                    <th>Thu</th>\n";
    echo "                    <th>Fri</th>\n";
    echo "                    <th class='sat-header'>Sat</th>\n";
    echo "                </tr>\n";
    
    $firstDayTimestamp = strtotime("$currentYear-$currentMonth-01");
    $firstDayIndex     = (int)date('w', $firstDayTimestamp); 
    $totalDays         = (int)date('t', $firstDayTimestamp); 
    
    $dateCounter = 1;
    $liveYear    = (int)date('Y');
    $liveMonth   = (int)date('n');
    $liveDay     = (int)date('j');

    for ($rowIdx = 0; $rowIdx < 6; $rowIdx++) {
        if ($dateCounter > $totalDays) { break; }
        
        echo "  <tr>\n";
        
        
        // Calculate week number based strictly on the Sunday date of this row
        if ($rowIdx === 0) {
            // In the first row, find the actual date of the first Sunday (even if it's in the last month)
            $rowSundayTimestamp = strtotime("$currentYear-$currentMonth-01 -" . $firstDayIndex . " days");
        } else {
            // For later rows, calculate directly from the current dateCounter day
            $rowSundayTimestamp = strtotime("$currentYear-$currentMonth-$dateCounter");
        }
        
        // Get standard calendar week number cleanly
        $weekNumber = (int)date('W', $rowSundayTimestamp);
        
        // Format week output with a leading zero if less than 10
        $displayWeek = ($weekNumber < 10) ? "0" . $weekNumber : $weekNumber;
        
        echo "                    <td class='wk-cell'>{$displayWeek}</td>\n";
        
        for ($colIdx = 0; $colIdx < 7; $colIdx++) {
            if (($rowIdx === 0 && $colIdx < $firstDayIndex) || $dateCounter > $totalDays) {
                echo "                    <td class='empty-cell'></td>\n";
            } else {
                $classes = ['day-cell'];
                if ($colIdx === 0) $classes[] = 'sun-cell';
                if ($colIdx === 6) $classes[] = 'sat-cell';
                
                if ($currentYear === $liveYear && $currentMonth === $liveMonth && $dateCounter === $liveDay) {
                    $classes[] = 'today-bold';
                }
                if ($selectedDay === $dateCounter) {
                    $classes[] = 'selected-blue';
                }
                
                $classStr = implode(' ', $classes);
                
                echo "                    <td class='{$classStr}'>\n";
                echo "                        <button type='submit' name='select_day' value='{$dateCounter}' data-info='Day {$dateCounter} cell'>{$dateCounter}</button>\n";
                echo "                    </td>\n";
                
                $dateCounter++;
            }
        }
        echo "                </tr>\n";
    }
    
    echo "                <tr class='row-footer'>\n";
    echo "                    <td colspan='8' class='calendar-footer'>{$footerText}</td>\n";
    echo "                </tr>\n";
    echo "            </table>\n";
    echo "        </form>\n";
    echo "    </div>\n";
}

echo "</div>\n";
echo "</body>\n";
echo "</html>\n";
?>
