<?php

class Auth
{
    // Start session if not already started
    public static function startSession()
    {
        if (session_status() == PHP_SESSION_NONE)
        {
            session_start();
        }
    }

    // Check whether user is logged in
    public static function checkLogin()
    {
        self::startSession();

        if (!isset($_SESSION['user_id']))
        {
            header("Location:index.php?action=login");
            exit;
        }
    }

    // Check user role
    public static function checkRole($role)
    {
        self::checkLogin();

        if ($_SESSION['role'] != $role)
        {
            die("Access Denied");
        }
    }

    // Get current logged-in user
    public static function user()
    {
        self::checkLogin();

        return [
            "id"   => $_SESSION['user_id'],
            "name" => $_SESSION['full_name'],
            "role" => $_SESSION['role']
        ];
    }

    // Logout
    public static function logout()
    {
        self::startSession();

        session_destroy();

        header("Location:index.php?action=login");
        exit;
    }
}

<?php

require_once "../app/core/Controller.php";
require_once "../app/core/Auth.php";
require_once "../app/models/Form.php";
require_once "../app/models/Submission.php";
require_once "../app/models/User.php";

class DashboardController extends Controller
{
    private $form;
    private $submission;
    private $user;

    public function __construct()
    {
        parent::__construct();

        $this->form = new Form();
        $this->submission = new Submission();
        $this->user = new User();
    }

    // ===========================
    // User Dashboard
    // ===========================
    public function userDashboard()
    {
        $this->requireRole("user");

        $user = Auth::user();

        $assignedForms = $this->form->getAssignedForms($user['id']);

        $submittedForms = $this->submission
                               ->getUserSubmissions($user['id']);

        $this->smarty->assign("user",$user);

        $this->smarty->assign(
            "assignedForms",
            $assignedForms
        );

        $this->smarty->assign(
            "submittedForms",
            $submittedForms
        );

        $this->smarty->display("dashboard.tpl");
    }

    // ===========================
    // Super Admin Dashboard
    // ===========================
    public function adminDashboard()
    {
        $this->requireRole("super_admin");

        $user = Auth::user();

        $users = $this->user->getAllUsers();

        $forms = $this->form->getForms();

        $submissions =
        $this->submission->getAllSubmissions();

        $this->smarty->assign("user",$user);

        $this->smarty->assign("users",$users);

        $this->smarty->assign("forms",$forms);

        $this->smarty->assign(
            "submissions",
            $submissions
        );

        $this->smarty->display(
            "admin_dashboard.tpl"
        );
    }
}

public function getAllUsers()
{
    $stmt = $this->conn->prepare(
        "SELECT *
         FROM users
         WHERE role='user'
         ORDER BY full_name"
    );

    $stmt->execute();

    return $stmt->get_result()->fetch_all(MYSQLI_ASSOC);
}

public function getForms()
{
    $stmt = $this->conn->prepare(
        "SELECT *
         FROM forms
         ORDER BY form_name"
    );

    $stmt->execute();

    return $stmt->get_result()->fetch_all(MYSQLI_ASSOC);
}
public function getAssignedForms($userId)
{
    $stmt = $this->conn->prepare(

        "SELECT
            f.*

        FROM form_access fa

        INNER JOIN forms f

        ON f.id = fa.form_id

        WHERE fa.user_id=?"

    );

    $stmt->bind_param("i",$userId);

    $stmt->execute();

    return $stmt->get_result()->fetch_all(MYSQLI_ASSOC);
}
