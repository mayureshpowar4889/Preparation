
//    VALID EMAIL


let objValidEmail = document.getElementsByClassName("valid-email");
for (let intLoopCount = 0; intLoopCount < objValidEmail.length; intLoopCount++) {

    objValidEmail[intLoopCount].addEventListener("blur", function () {
        let strValue = this.value.trim();
        let strLabel = document.querySelector('label[for="' + this.name + '"]').innerText;
        let regEmail = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        if (strValue != "" && !regEmail.test(strValue)) {
            alert("Invalid " + strLabel);
        }
    }
    );
}

//    VALID MIN LENGTH

let objMinLength = document.querySelectorAll(
    '[class*="valid-min-"]');
for (let intLoopCount = 0; intLoopCount < objMinLength.length; intLoopCount++
) {

    objMinLength[intLoopCount].addEventListener("blur",
        function () {

            let strValue = this.value.trim();
            let strLabel = document.querySelector(
                'label[for="' +
                this.name +
                '"]').innerText;
            this.classList.forEach(function (strClass) {
                if (strClass.startsWith("valid-min-")) {

                    let intMin = Number(strClass.split("-")[2]);
                    if (strValue.length < intMin) {
                        alert(strLabel + " minimum length is " + intMin);

                    }
                }
            });

        }

    );

}
