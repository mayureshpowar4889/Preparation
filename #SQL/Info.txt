//Use for Attach Solution and Attch Note
function hideRequiredattch() {

    let objGetValue = document.getElementsByName("attach");
    let objReqa = document.getElementById("attachid");
    let objReqb = document.getElementById("note");
    let objSpan = document.getElementById("attsol");

    for (let intLoopCount = 0; intLoopCount < objGetValue.length; intLoopCount++) {
        if (objGetValue[intLoopCount].checked) {
            if (objGetValue[intLoopCount].value == "1") {
                objReqa.style.display = '';
            } else {

                objReqb.style.display = 'inline';
                objSpan.style.display = 'inline';
            } else {
                objReqb.style.display = 'none';
                objSpan.style.display = 'none';
            }
        }
    }
}

//Use for file Valid
function fileValid() {
    let intUploadFileSize = document.getElementById("file").files[0].size;
    let size = 2097152;
    if (intUploadFileSize >= size) {
        alert("File size is too big - limit 2MB")
        document.getElementById("file").value = '';
    }
}


//Use for Add more files
function addfile() {

    let intCountOfFile = document.getElementById("addfile").childElementCount;

    if (intCountOfFile >= 10) {
        alert("Only 10 files Limit");
    }
    else {
        let objDiv = document.createElement("div");
        objDiv.className = "filebox";
        let objInput = document.createElement('input');
        objInput.type = "file";
        let objRemovebtn = document.createElement("button");
        objRemovebtn.innerText = "Remove";
        objRemovebtn.className = "remBtn";
        objRemovebtn.onclick = function () {
            objDiv.remove();
        }
        objDiv.appendChild(objInput);
        objDiv.appendChild(objRemovebtn);
        document.getElementById("addfile").appendChild(objDiv);
    }

}

//date check
function validateDate(input) {

    let value = input.value.replace(/\D/g, "");

    value = value.slice(0, 8);

    if (value.length > 2 && value.length <= 4) {
        value = value.slice(0, 2) + "-" + value.slice(2);
    }
    else if (value.length > 4) {
        value = value.slice(0, 2) + "-" + value.slice(2, 4) + "-" + value.slice(4, 8);
    }

    input.value = value;

    if (value.length == 10) {
        let arrDate = value.split("-");
        let intDate = Number(arrDate[0]);
        let intMonth = Number(arrDate[1]);
        let intYear = Number(arrDate[2]);

        if (intDate < 1 ||
            intDate > 31) {

            alert(" Enter date between 1 to 31");
            input.value = "";
            return;
        }
        if (intMonth < 1 || intMonth > 12) {

            alert("Month must be between 1 to 12");


            input.value = "";
            return;
        }


        if (arrDate[2].length != 4) {

            alert("Year must be 4 digits");

            input.value = "";
            return;
        }

    }

}


//used for time check
function validateTime(input) {

    let value = input.value.replace(/\D/g, "");
    value = value.slice(0, 4);

    if (value.length > 2) {
        value = value.slice(0, 2) + ":" + value.slice(2, 4);
    }

    input.value = value;

    if (value.length == 5) {

        let arrTime = value.split(":");
        let intHour = Number(arrTime[0]);

        let intMinute = Number(arrTime[1]);


        if (intHour < 1 ||
            intHour > 12) {
            alert("Hour between 1 to 12");
            input.value = "";
            return;
        }
        if (intMinute < 0 ||
            intMinute > 59) {
            alert("Minute between 0 to 59");
            input.value = "";
            return;
        }

    }

    calculateDuration();

}

function calculateDuration() {

    const objStartTime = document.querySelectorAll(".time")[1];
    const objStartAMPM = document.querySelectorAll(".select-g")[2];
    const objEndTime = document.querySelectorAll(".time")[2];
    const objEndAMPM = document.querySelectorAll(".select-g")[3];


    let arrStart = objStartTime.value.split(":");
    let intStartHour = Number(arrStart[0]);
    let intStartMinute = Number(arrStart[1]);



    let arrEnd = objEndTime.value.split(":");
    let intEndHour = Number(arrEnd[0]);
    let intEndMinute = Number(arrEnd[1]);



    if (objStartAMPM.value == "PM" && intStartHour != 12) {
        intStartHour = intStartHour + 12;
    }

    if (objStartAMPM.value == "AM" && intStartHour == 12) {
        intStartHour = 0;
    }
    if (objEndAMPM.value == "PM" && intEndHour != 12) {

        intEndHour = intEndHour + 12;
    }
    if (objEndAMPM.value == "AM" && intEndHour == 12) {
        intEndHour = 0;
    }


    let intStartTotal = (intStartHour * 60) + intStartMinute;
    let intEndTotal = (intEndHour *60) + intEndMinute;

    let intDura = intEndTotal - intStartTotal;

    let intHours = Math.floor(intDura/60);
    let intMinutes = intDura % 60;



 

    document.querySelectorAll(".size-a")[0].value = intHours;

    document.querySelectorAll(".size-a")[1].value = intMinutes;

}
