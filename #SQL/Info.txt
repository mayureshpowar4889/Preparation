document
.querySelectorAll(".slot")
.forEach(function(cell)
{
    cell.addEventListener(
        "mouseover",
        function()
        {
            if(!isDragging)
            {
                return;
            }

            dragEndCell = this;

            document
            .querySelectorAll(".dragging")
            .forEach(function(item)
            {
                item.classList.remove(
                    "dragging"
                );
            });

            let day =
            parseInt(
                dragStartCell.dataset.day
            );

            let currentDay =
            parseInt(
                this.dataset.day
            );

            if(
                day != currentDay
            )
            {
                return;
            }

            let startSlot =
            parseInt(
                dragStartCell.dataset.slot
            );

            let endSlot =
            parseInt(
                dragEndCell.dataset.slot
            );

            if(endSlot < startSlot)
            {
                let temp =
                startSlot;

                startSlot =
                endSlot;

                endSlot =
                temp;
            }

            document
            .querySelectorAll(".slot")
            .forEach(function(item)
            {
                if(
                    parseInt(item.dataset.day)
                    ==
                    day

                    &&

                    parseInt(item.dataset.slot)
                    >=
                    startSlot

                    &&

                    parseInt(item.dataset.slot)
                    <=
                    endSlot
                )
                {
                    item.classList.add(
                        "dragging"
                    );
                }
            });
        }
    );
});
