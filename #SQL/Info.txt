<?php

require_once "../app/core/Model.php";

class Access extends Model
{

    // Get All Users
    public function getUsers()
    {
        $stmt = $this->conn->prepare(

            "SELECT
                id,
                full_name,
                email

             FROM users

             WHERE role='user'

             ORDER BY full_name"

        );

        $stmt->execute();

        return $stmt->get_result()->fetch_all(MYSQLI_ASSOC);
    }

    // Get All Forms
    public function getForms()
    {
        $stmt = $this->conn->prepare(

            "SELECT *

             FROM forms

             ORDER BY form_name"

        );

        $stmt->execute();

        return $stmt->get_result()->fetch_all(MYSQLI_ASSOC);
    }

    // Forms Assigned To User
    public function getUserForms($userId)
    {
        $stmt = $this->conn->prepare(

            "SELECT

                f.*

             FROM form_access fa

             INNER JOIN forms f

             ON f.id = fa.form_id

             WHERE fa.user_id=?"

        );

        $stmt->bind_param("i",$userId);

        $stmt->execute();

        return $stmt->get_result()->fetch_all(MYSQLI_ASSOC);
    }

    // Grant Form Access
    public function grantAccess(
        $userId,
        $formId,
        $grantedBy
    )
    {
        $stmt = $this->conn->prepare(

            "INSERT INTO form_access
            (
                user_id,
                form_id,
                granted_by
            )

            VALUES
            (
                ?,?,?
            )"

        );

        $stmt->bind_param(

            "iii",

            $userId,

            $formId,

            $grantedBy

        );

        return $stmt->execute();
    }

    // Remove Form Access
    public function removeAccess(
        $userId,
        $formId
    )
    {
        $stmt = $this->conn->prepare(

            "DELETE FROM form_access

             WHERE user_id=?

             AND form_id=?"

        );

        $stmt->bind_param(

            "ii",

            $userId,

            $formId

        );

        return $stmt->execute();
    }

    // Check if access already exists
    public function hasAccess(
        $userId,
        $formId
    )
    {
        $stmt = $this->conn->prepare(

            "SELECT id

             FROM form_access

             WHERE user_id=?

             AND form_id=?"

        );

        $stmt->bind_param(

            "ii",

            $userId,

            $formId

        );

        $stmt->execute();

        return $stmt
               ->get_result()
               ->num_rows > 0;
    }

}

<?php

require_once "../app/core/Controller.php";
require_once "../app/core/Auth.php";

require_once "../app/models/Access.php";
require_once "../app/models/Form.php";

class AdminController extends Controller
{
    private $access;
    private $form;

    public function __construct()
    {
        parent::__construct();

        $this->requireRole("super_admin");

        $this->access = new Access();

        $this->form = new Form();
    }

    // ==========================
    // Manage Form Access Page
    // ==========================
    public function manageAccess()
    {
        $users = $this->access->getUsers();

        $selectedUser = "";

        $assignedForms = [];

        $availableForms = [];

        if(isset($_GET['user_id']))
        {
            $selectedUser = $_GET['user_id'];

            $assignedForms =
            $this->access->getUserForms($selectedUser);

            $allForms =
            $this->form->getForms();

            $assignedIds =
            array_column($assignedForms,'id');

            foreach($allForms as $form)
            {
                if(!in_array($form['id'],$assignedIds))
                {
                    $availableForms[] = $form;
                }
            }
        }

        $this->smarty->assign(
            "users",
            $users
        );

        $this->smarty->assign(
            "selectedUser",
            $selectedUser
        );

        $this->smarty->assign(
            "assignedForms",
            $assignedForms
        );

        $this->smarty->assign(
            "availableForms",
            $availableForms
        );

        $this->smarty->display(
            "manage_access.tpl"
        );
    }

    // ==========================
    // Grant Access
    // ==========================
    public function grantAccess()
    {
        $userId = $_POST['user_id'];

        $formId = $_POST['form_id'];

        $admin = Auth::user();

        if(
            !$this->access
                  ->hasAccess($userId,$formId)
        )
        {
            $this->access->grantAccess(

                $userId,

                $formId,

                $admin['id']

            );
        }

        header(

            "Location:index.php?action=manageAccess&user_id=".$userId

        );

        exit;
    }

    // ==========================
    // Remove Access
    // ==========================
    public function removeAccess()
    {
        $userId = $_GET['user_id'];

        $formId = $_GET['form_id'];

        $this->access->removeAccess(

            $userId,

            $formId

        );

        header(

            "Location:index.php?action=manageAccess&user_id=".$userId

        );

        exit;
    }

}
