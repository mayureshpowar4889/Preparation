<?php

/* ==========================================
   ONLY POST REQUEST ALLOWED
========================================== */

if ($_SERVER["REQUEST_METHOD"] != "POST")
{
    die("Invalid Request");
}


/* ==========================================
   GET FORM DATA
========================================== */

$name = trim($_POST["name"] ?? "");

$email = trim($_POST["email"] ?? "");

$workEmail = trim($_POST["work_email"] ?? "");

$dob = trim($_POST["dob"] ?? "");

$gender = trim($_POST["gender"] ?? "");

$state = trim($_POST["state"] ?? "");

$city = trim($_POST["city"] ?? "");

$password = $_POST["password"] ?? "";

$confirmPassword =
$_POST["confirm_password"] ?? "";

$terms =
isset($_POST["terms"]);


/* ==========================================
   BACKEND VALIDATION
========================================== */

$errors = [];


/* NAME */

if(empty($name))
{
    $errors[] =
    "Name is required";
}
elseif(
    !preg_match(
    "/^[A-Za-z ]+$/",
    $name
    )
)
{
    $errors[] =
    "Name should contain only alphabets";
}
elseif(strlen($name) < 3)
{
    $errors[] =
    "Name must contain minimum 3 characters";
}


/* PERSONAL EMAIL */

if(empty($email))
{
    $errors[] =
    "Personal email is required";
}
elseif(
    !filter_var(
    $email,
    FILTER_VALIDATE_EMAIL
    )
)
{
    $errors[] =
    "Invalid personal email";
}


/* WORK EMAIL */

if(empty($workEmail))
{
    $errors[] =
    "Work email is required";
}
elseif(
    !filter_var(
    $workEmail,
    FILTER_VALIDATE_EMAIL
    )
)
{
    $errors[] =
    "Invalid work email";
}


/* DOB */

if(empty($dob))
{
    $errors[] =
    "Date of birth is required";
}
else
{
    $birthDate =
    new DateTime($dob);

    $today =
    new DateTime();

    $age =
    $today->diff($birthDate)->y;

    if($age < 18)
    {
        $errors[] =
        "Age must be 18 or above";
    }
}


/* GENDER */

if(empty($gender))
{
    $errors[] =
    "Gender is required";
}


/* STATE */

if(empty($state))
{
    $errors[] =
    "State is required";
}


/* CITY */

if(empty($city))
{
    $errors[] =
    "City is required";
}


/* PASSWORD */

$passwordPattern =
"/^(?=.*[a-z])(?=.*[A-Z])(?=.*\d)(?=.*[@$!%*?&]).{8,}$/";

if(empty($password))
{
    $errors[] =
    "Password is required";
}
elseif(
    !preg_match(
    $passwordPattern,
    $password
    )
)
{
    $errors[] =
    "Password format invalid";
}


/* CONFIRM PASSWORD */

if($password !== $confirmPassword)
{
    $errors[] =
    "Passwords do not match";
}


/* TERMS */

if(!$terms)
{
    $errors[] =
    "Accept terms and conditions";
}


/* ==========================================
   READ JSON FILE
========================================== */

$fileName = "data.json";

$jsonData =
file_get_contents($fileName);

$users =
json_decode($jsonData, true);

if(!is_array($users))
{
    $users = [];
}


/* ==========================================
   DUPLICATE EMAIL CHECK
========================================== */

foreach($users as $user)
{
    if(
        strtolower($user["email"])
        ==
        strtolower($email)
    )
    {
        $errors[] =
        "Personal email already exists";

        break;
    }
}


/* ==========================================
   SHOW ERRORS
========================================== */

if(count($errors) > 0)
{
    echo "<h2>Validation Errors</h2>";

    echo "<ul>";

    foreach($errors as $error)
    {
        echo "<li>" .
             htmlspecialchars($error)
             .
             "</li>";
    }

    echo "</ul>";

    echo '<a href="index.php">
          Back
          </a>';

    exit;
}


/* ==========================================
   HASH PASSWORD
========================================== */

$hashedPassword =
password_hash(
$password,
PASSWORD_DEFAULT
);


/* ==========================================
   CREATE USER ARRAY
========================================== */

$newUser =
[
    "name" => $name,

    "email" => $email,

    "work_email" => $workEmail,

    "dob" => $dob,

    "gender" => $gender,

    "state" => $state,

    "city" => $city,

    "password" =>
    $hashedPassword,

    "created_at" =>
    date("Y-m-d H:i:s")
];


/* ==========================================
   APPEND NEW USER
========================================== */

$users[] =
$newUser;


/* ==========================================
   SAVE JSON
========================================== */

file_put_contents(
    $fileName,
    json_encode(
        $users,
        JSON_PRETTY_PRINT
    )
);


/* ==========================================
   SUCCESS MESSAGE
========================================== */

echo "
<!DOCTYPE html>
<html>
<head>
<title>Success</title>
</head>
<body>

<h2>
Registration Successful
</h2>

<p>
Welcome <b>"
. htmlspecialchars($name) .
"</b>
</p>

<p>
Your data has been stored successfully.
</p>

<a href='index.php'>
Register Another User
</a>

</body>
</html>
";
?>
