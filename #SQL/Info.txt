/* =========================================
   SLOT TO TIME
========================================= */

function slotToTime(slot)
{
    let hour =
    Math.floor(
        slot / 2
    );

    let minute =
    (slot % 2)
    *
    30;

    let ampm =
    hour >= 12
    ?
    "PM"
    :
    "AM";

    let displayHour =
    hour % 12;

    if(displayHour == 0)
    {
        displayHour = 12;
    }

    return (
        displayHour
        +
        ":"
        +
        minute
        .toString()
        .padStart(
            2,
            "0"
        )
        +
        ampm
    );
}

function renderMeetings()
{
    meetings.forEach(
    function(meeting)
    {
        let startCell =
        document.querySelector(
        ".slot[data-day='"
        +
        meeting.day
        +
        "'][data-slot='"
        +
        meeting.start
        +
        "']"
        );

        if(
            !startCell
        )
        {
            return;
        }

        let box =
        document.createElement(
            "div"
        );

        box.className =
        "meeting-box";

        let totalSlots =
        meeting.end
        -
        meeting.start
        +
        1;

        box.style.height =
        (
            totalSlots
            *
            25
            -
            4
        )
        +
        "px";

        box.innerHTML =
        "<b>"
        +
        slotToTime(
            meeting.start
        )
        +
        " - "
        +
        slotToTime(
            meeting.end
        )
        +
        "</b>"
        +
        "<br>"
        +
        meeting.title;

        startCell.appendChild(
            box
        );
    });
}
