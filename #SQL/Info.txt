    <script>





        /* =========================================
           GET ALL SUBMIT BUTTONS
        ========================================== */

        const objButtons =
        document.querySelectorAll(".submit-btn");





        /* =========================================
           LOOP BUTTONS
        ========================================== */

        objButtons.forEach(function(objBtn){

            objBtn.addEventListener(

                "click",

                validateForm

            );

        });





        /* =========================================
           MAIN VALIDATION
        ========================================== */

        function validateForm(event){




            /* =========================================
               CURRENT FORM
            ========================================== */

            let objForm =

            event.target.closest("form");




            /* =========================================
               ALL FIELDS
            ========================================== */

            let objFields =

            objForm.querySelectorAll(

                "input, textarea, select"

            );




            let arrError = [];





            /* =========================================
               LOOP FIELDS
            ========================================== */

            objFields.forEach(function(objField){




                /* =========================================
                   LABEL
                ========================================== */

                let strLabel =

                objForm
                .querySelector(

                    'label[for="' +
                    objField.id +
                    '"]'

                )
                .innerText;





                /* =========================================
                   VALUE
                ========================================== */

                let strValue =

                objField.value.trim();





                /* =========================================
                   REQUIRED
                ========================================== */

                if(

                    objField.required &&
                    strValue == ""

                ){

                    arrError.push(

                        strLabel +
                        " is required"

                    );

                    return;

                }





                /* =========================================
                   MIN LENGTH
                ========================================== */

                if(

                    objField.minLength > 0 &&
                    strValue.length <
                    objField.minLength

                ){

                    arrError.push(

                        strLabel +
                        " minimum length is " +
                        objField.minLength

                    );

                }





                /* =========================================
                   MAX LENGTH
                ========================================== */

                if(

                    objField.maxLength > 0 &&
                    strValue.length >
                    objField.maxLength

                ){

                    arrError.push(

                        strLabel +
                        " maximum length is " +
                        objField.maxLength

                    );

                }





                /* =========================================
                   EMAIL
                ========================================== */

                if(

                    objField.dataset.type ==
                    "email"

                ){

                    let regEmail =

                    /^[^\s@]+@[^\s@]+\.[^\s@]+$/;



                    if(

                        strValue != "" &&
                        !regEmail.test(strValue)

                    ){

                        arrError.push(

                            "Invalid " +
                            strLabel

                        );

                    }

                }





                /* =========================================
                   PHONE
                ========================================== */

                if(

                    objField.dataset.type ==
                    "phone"

                ){

                    let regPhone =
                    /^[0-9]{10}$/;



                    if(

                        strValue != "" &&
                        !regPhone.test(strValue)

                    ){

                        arrError.push(

                            "Invalid " +
                            strLabel

                        );

                    }

                }





                /* =========================================
                   URL
                ========================================== */

                if(

                    objField.dataset.type ==
                    "url"

                ){

                    let regUrl =

                    /^(https?:\/\/)?([\w\-])+\.{1}[a-zA-Z]{2,}(\/.*)?$/;



                    if(

                        strValue != "" &&
                        !regUrl.test(strValue)

                    ){

                        arrError.push(

                            "Invalid " +
                            strLabel

                        );

                    }

                }





                /* =========================================
                   NO NUMBER
                ========================================== */

                if(

                    objField.dataset.nonumber ==
                    "true"

                ){

                    let regNumber = /\d/;



                    if(

                        regNumber.test(strValue)

                    ){

                        arrError.push(

                            strLabel +
                            " should not contain numbers"

                        );

                    }

                }





                /* =========================================
                   NO SPECIAL CHARACTER
                ========================================== */

                if(

                    objField.dataset.nochar ==
                    "true"

                ){

                    let regChar =
                    /[^a-zA-Z0-9]/;



                    if(

                        regChar.test(strValue)

                    ){

                        arrError.push(

                            strLabel +
                            " should not contain special characters"

                        );

                    }

                }





                /* =========================================
                   DATE
                ========================================== */

                if(

                    objField.dataset.type ==
                    "date"

                ){

                    let regDate =
                    /^\d{2}-\d{2}-\d{4}$/;



                    if(

                        strValue != "" &&
                        !regDate.test(strValue)

                    ){

                        arrError.push(

                            "Invalid " +
                            strLabel

                        );

                    }

                }





                /* =========================================
                   TIME
                ========================================== */

                if(

                    objField.dataset.type ==
                    "time"

                ){

                    let regTime =
                    /^(0[1-9]|1[0-2]):([0-5][0-9])$/;



                    if(

                        strValue != "" &&
                        !regTime.test(strValue)

                    ){

                        arrError.push(

                            "Invalid " +
                            strLabel

                        );

                    }

                }

            });





            /* =========================================
               SHOW ERRORS
            ========================================== */

            if(arrError.length > 0){

                alert(

                    arrError.join("\n")

                );

            }

            else{

                alert(

                    "Form Submitted Successfully"

                );

            }

        }
