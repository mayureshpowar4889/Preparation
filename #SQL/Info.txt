<?php

class MedTemplate
{
    private PDO $objDatabase;

    public function __construct(PDO $objDatabase)
    {
        $this->objDatabase = $objDatabase;
    }

    /*
    |--------------------------------------------------------------------------
    | Save Template
    |--------------------------------------------------------------------------
    */

    public function saveTemplate(array $arrTemplateData): bool
    {
        $strQuery = "
            INSERT INTO templates
            (
                title,
                subject,
                description,
                created_by
            )
            VALUES
            (
                :title,
                :subject,
                :description,
                :created_by
            )
        ";

        $objStatement = $this->objDatabase->prepare($strQuery);

        return $objStatement->execute([
            ':title'       => $arrTemplateData['title'],
            ':subject'     => $arrTemplateData['subject'],
            ':description' => $arrTemplateData['description'],
            ':created_by'  => $arrTemplateData['created_by']
        ]);
    }

    /*
    |--------------------------------------------------------------------------
    | Get All Templates
    |--------------------------------------------------------------------------
    */

    public function getAllTemplates(): array
    {
        $strQuery = "
            SELECT
                id,
                title,
                subject
            FROM templates
            ORDER BY id DESC
        ";

        $objStatement = $this->objDatabase->prepare($strQuery);

        $objStatement->execute();

        return $objStatement->fetchAll(PDO::FETCH_ASSOC);
    }

    /*
    |--------------------------------------------------------------------------
    | Get Template By ID
    |--------------------------------------------------------------------------
    */

    public function getTemplateById(int $intTemplateId): array|false
    {
        $strQuery = "
            SELECT
                *
            FROM templates
            WHERE id = :id
        ";

        $objStatement = $this->objDatabase->prepare($strQuery);

        $objStatement->execute([
            ':id' => $intTemplateId
        ]);

        return $objStatement->fetch(PDO::FETCH_ASSOC);
    }
}

public function save()
{
    if ($_SERVER['REQUEST_METHOD'] != 'POST') {
        header("Location: index.php?controller=Template&action=create");
        exit;
    }

    $arrTemplateData = [];

    $arrTemplateData['title'] = trim($_POST['title']);
    $arrTemplateData['subject'] = trim($_POST['subject']);
    $arrTemplateData['description'] = $_POST['description'];
    $arrTemplateData['created_by'] = $_SESSION['user_id'];

    if (
        empty($arrTemplateData['title']) ||
        empty($arrTemplateData['subject']) ||
        empty($arrTemplateData['description'])
    ) {
        $_SESSION['error'] = "All fields are required.";

        header("Location: index.php?controller=Template&action=create");
        exit;
    }

    $blnResult = $this->objMedTemplate->saveTemplate($arrTemplateData);

    if ($blnResult) {

        $_SESSION['success'] = "Template created successfully.";

    } else {

        $_SESSION['error'] = "Unable to save template.";

    }

    header("Location: index.php?controller=Template&action=create");
    exit;
}
