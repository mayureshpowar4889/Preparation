<?php

require_once "../app/core/Model.php";

class MedFormAccess extends Model
{

    /**
     * --------------------------------------------------------
     * Get All Users
     * --------------------------------------------------------
     */
    public function getAllUsers()
    {

        $strQuery = "

            SELECT

                id,

                full_name,

                email

            FROM users

            WHERE role='user'

            ORDER BY full_name

        ";

        $objStatement =
        $this->conn->prepare($strQuery);

        $objStatement->execute();

        $arrAllUsers =
        $objStatement
        ->get_result()
        ->fetch_all(MYSQLI_ASSOC);

        return $arrAllUsers;

    }

    /**
     * --------------------------------------------------------
     * Get All Forms
     * --------------------------------------------------------
     */
    public function getAllForms()
    {

        $strQuery = "

            SELECT

                id,

                form_name

            FROM forms

            ORDER BY form_name

        ";

        $objStatement =
        $this->conn->prepare($strQuery);

        $objStatement->execute();

        $arrAllForms =
        $objStatement
        ->get_result()
        ->fetch_all(MYSQLI_ASSOC);

        return $arrAllForms;

    }

    /**
     * --------------------------------------------------------
     * Manage Form Access List
     * --------------------------------------------------------
     *
     * Returns
     *
     * User
     * Form
     * Status
     * Grant/Revoke
     *
     * --------------------------------------------------------
     */

    public function getAccessList()
    {

        $strQuery = "

            SELECT

                u.id AS user_id,

                u.full_name,

                u.email,

                f.id AS form_id,

                f.form_name,

                fa.id AS access_id,

                CASE

                    WHEN fa.id IS NULL

                    THEN 'Not Granted'

                    ELSE 'Granted'

                END AS access_status

            FROM users u

            CROSS JOIN forms f

            LEFT JOIN form_access fa

                ON fa.user_id = u.id

               AND fa.form_id = f.id

            WHERE u.role='user'

            ORDER BY

                u.full_name,

                f.form_name

        ";

        $objStatement =
        $this->conn->prepare($strQuery);

        $objStatement->execute();

        $arrAccessList =
        $objStatement
        ->get_result()
        ->fetch_all(MYSQLI_ASSOC);

        return $arrAccessList;

    }
/**
     * --------------------------------------------------------
     * Check User Form Access
     * --------------------------------------------------------
     */
    public function hasAccess(
        $intUserId,
        $intFormId
    )
    {

        $strQuery = "

            SELECT

                id

            FROM form_access

            WHERE user_id = ?

            AND form_id = ?

        ";

        $objStatement =
        $this->conn->prepare($strQuery);

        $objStatement->bind_param(

            "ii",

            $intUserId,

            $intFormId

        );

        $objStatement->execute();

        $objResult =
        $objStatement->get_result();

        $blStatus =
        ($objResult->num_rows > 0);

        return $blStatus;

    }

    /**
     * --------------------------------------------------------
     * Grant Form Access
     * --------------------------------------------------------
     */
    public function grantAccess(
        $intUserId,
        $intFormId,
        $intGrantedBy
    )
    {

        if(
            $this->hasAccess(
                $intUserId,
                $intFormId
            )
        )
        {
            return false;
        }

        $strQuery = "

            INSERT INTO form_access
            (
                user_id,
                form_id,
                granted_by
            )

            VALUES
            (
                ?, ?, ?
            )

        ";

        $objStatement =
        $this->conn->prepare($strQuery);

        $objStatement->bind_param(

            "iii",

            $intUserId,

            $intFormId,

            $intGrantedBy

        );

        $blStatus =
        $objStatement->execute();

        return $blStatus;

    }

    /**
     * --------------------------------------------------------
     * Revoke Form Access
     * --------------------------------------------------------
     */
    public function revokeAccess(
        $intUserId,
        $intFormId
    )
    {

        $strQuery = "

            DELETE

            FROM form_access

            WHERE user_id = ?

            AND form_id = ?

        ";

        $objStatement =
        $this->conn->prepare($strQuery);

        $objStatement->bind_param(

            "ii",

            $intUserId,

            $intFormId

        );

        $blStatus =
        $objStatement->execute();

        return $blStatus;

    }

}

<?php

require_once "../app/core/Controller.php";
require_once "../app/core/Auth.php";

require_once "../app/models/med_form_access.php";

class MedAdminController extends Controller
{
    private $objMedFormAccess;

    public function __construct()
    {
        parent::__construct();

        $this->requireRole("super_admin");

        $this->objMedFormAccess =
        new MedFormAccess();
    }

    /**
     * ------------------------------------------------
     * Manage Form Access
     * ------------------------------------------------
     */
    public function manageFormAccess()
    {
        $arrAllUsers =
        $this->objMedFormAccess
             ->getAllUsers();

        $arrAllForms =
        $this->objMedFormAccess
             ->getAllForms();

        $arrAccessList =
        $this->objMedFormAccess
             ->getAccessList();

        $this->smarty->assign(
            "arrAllUsers",
            $arrAllUsers
        );

        $this->smarty->assign(
            "arrAllForms",
            $arrAllForms
        );

        $this->smarty->assign(
            "arrAccessList",
            $arrAccessList
        );

        $this->smarty->display(
            "manage_form_access.tpl"
        );
    }

    /**
     * ------------------------------------------------
     * Grant Form Access
     * ------------------------------------------------
     */
    public function grantFormAccess()
    {
        if($_SERVER['REQUEST_METHOD']!="POST")
        {
            header(
                "Location:index.php?action=manageFormAccess"
            );

            exit;
        }

        $intUserId =
        (int)$_POST['user_id'];

        $intFormId =
        (int)$_POST['form_id'];

        $arrLoggedInUser =
        Auth::user();

        $intGrantedBy =
        $arrLoggedInUser['id'];

        $blStatus =
        $this->objMedFormAccess
             ->grantAccess(
                $intUserId,
                $intFormId,
                $intGrantedBy
             );

        header(
            "Location:index.php?action=manageFormAccess"
        );

        exit;
    }
/**
     * ------------------------------------------------
     * Revoke Form Access
     * ------------------------------------------------
     */
    public function revokeFormAccess()
    {

        if($_SERVER['REQUEST_METHOD']!="POST")
        {
            header(
                "Location:index.php?action=manageFormAccess"
            );

            exit;
        }

        $intUserId =
        (int)$_POST['user_id'];

        $intFormId =
        (int)$_POST['form_id'];

        $blStatus =
        $this->objMedFormAccess
             ->revokeAccess(
                $intUserId,
                $intFormId
             );

        header(
            "Location:index.php?action=manageFormAccess"
        );

        exit;

    }

}
